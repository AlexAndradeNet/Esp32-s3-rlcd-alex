// ================= SHTC3 SENSOR DRIVER (ESP-IDF I2C MASTER) =================
static i2c_master_dev_handle_t shtc3_dev_handle = NULL;

uint8_t crc8(const uint8_t *data, int len) { 
  uint8_t crc = 0xFF; 
  for (int i = 0; i < len; i++) { 
    crc ^= data[i]; 
    for (int b = 0; b < 8; b++) { 
      crc = (crc & 0x80) ? (uint8_t)((crc << 1) ^ 0x31) : (uint8_t)(crc << 1); 
    } 
  } 
  return crc; 
}

bool shtc3_read_i2cbus(float &tempC, float &rh) {
  if (shtc3_dev_handle == NULL) {
    i2c_device_config_t dev_cfg = {
      .dev_addr_length = I2C_ADDR_BIT_LEN_7,
      .device_address = SHTC3_ADDR,
      .scl_speed_hz = 100000,
    };
    if (i2c_master_bus_add_device(I2cbus.user_i2c_handle, &dev_cfg, &shtc3_dev_handle) != ESP_OK) {
      return false;
    }
  }

  // 1. Wakeup command (0x3517)
  uint8_t cmd_wake[2] = {0x35, 0x17};
  if (i2c_master_transmit(shtc3_dev_handle, cmd_wake, 2, 50) != ESP_OK) return false;
  delay(1);

  // 2. Measure command (normal power, clock stretching disabled: 0x7866)
  uint8_t cmd_meas[2] = {0x78, 0x66};
  if (i2c_master_transmit(shtc3_dev_handle, cmd_meas, 2, 50) != ESP_OK) return false;
  delay(20);

  // 3. Read 6 raw data bytes
  uint8_t d[6] = {0};
  if (i2c_master_receive(shtc3_dev_handle, d, 6, 50) != ESP_OK) return false;

  // 4. Verify CRC checksums
  if (crc8(&d[0], 2) != d[2]) return false;
  if (crc8(&d[3], 2) != d[5]) return false;

  uint16_t tRaw  = (uint16_t)d[0] << 8 | d[1];
  uint16_t rhRaw = (uint16_t)d[3] << 8 | d[4];

  tempC = -45.0f + 175.0f * (float)tRaw / 65535.0f;
  rh    = 100.0f * (float)rhRaw / 65535.0f;

  // 5. Sleep command (0xB098)
  uint8_t cmd_sleep[2] = {0xB0, 0x98};
  i2c_master_transmit(shtc3_dev_handle, cmd_sleep, 2, 50);

  return true;
}

  if (vbat >= 4.0) return 5;

}

void connectWiFi() {
  bool useFirst = true;
  while (WiFi.status() != WL_CONNECTED) {
    const char* ssid = useFirst ? ssid1 : ssid2;
    const char* pass = useFirst ? pass1 : pass2;

    Serial.printf("\nTrying %s\n", ssid);
    WiFi.begin(ssid, pass);

    unsigned long startAttemptTime = millis();
    while (WiFi.status() != WL_CONNECTED && (millis() - startAttemptTime < connectTimeout)) {
      delay(500);
      Serial.print(".");
    }

    if (WiFi.status() == WL_CONNECTED) {
      Serial.println("\nConnected!");
      return;
    }

    Serial.println("\nFailed. Switching network...");
    WiFi.disconnect(true);
    get_sdcard_config();
    delay(1000);
    useFirst = !useFirst;  
  }
}

void pushCanvasToRLCD(bool invert = false) {
  uint8_t *buf = canvas.getBuffer();        
  const int bytesPerRow = (W + 7) / 8;      

  RlcdPort.RLCD_ColorClear(ColorWhite);

  for (int y = 0; y < H; y++) {
    uint8_t *row = buf + y * bytesPerRow;
    for (int bx = 0; bx < bytesPerRow; bx++) {
      uint8_t v = row[bx];
      if (invert) v ^= 0xFF;

      int x0 = bx * 8;
      for (int bit = 0; bit < 8; bit++) {
        int x = x0 + bit;
        if (x >= W) break;

        bool on = (v & (0x80 >> bit)) != 0;
        if (on) {
          RlcdPort.RLCD_SetPixel((uint16_t)x, (uint16_t)y, ColorBlack);
        }
      }
    }
  }
  RlcdPort.RLCD_Display();
}

int ETA_in_mins(const char* eta) {
  if (eta == nullptr || strlen(eta) == 0) return -1;
  struct tm tm;
  memset(&tm, 0, sizeof(tm));
  strptime(eta, "%Y-%m-%dT%H:%M:%S", &tm);
  time_t busTime = mktime(&tm);
  time_t now;
  time(&now);
  return (int)(difftime(busTime, now) / 60);
}

bool syncTimeWithNTP() {
  configTime(atoi(gmt) * 3600, 0, "pool.ntp.org", "time.nist.gov");
  struct tm timeinfo;
  int retry = 0;
  while (!getLocalTime(&timeinfo) && retry < 15) {
    delay(500);
    retry++;
  }
  if (retry >= 15) return false;

  rtc.setDate(timeinfo.tm_wday, timeinfo.tm_mday, timeinfo.tm_mon + 1, timeinfo.tm_year + 1900);
  rtc.setTime(timeinfo.tm_hour, timeinfo.tm_min, timeinfo.tm_sec);
  return true;
}

void printWrapped(U8G2_FOR_ADAFRUIT_GFX &u8g2, const char *text, int startX, int startY, int maxWidth) {
  int cursorY = startY;
  int lineHeight = u8g2.getFontAscent() - u8g2.getFontDescent() + 4;
  char lineBuffer[256]; int lineLen = 0; int lineWidth = 0;
  char wordBuffer[128]; int wordLen = 0; int wordWidth = 0;

  int i = 0;
  while (text[i] != '\0') {
    unsigned char c = text[i];
    int charLen = (c < 0x80) ? 1 : ((c & 0xE0) == 0xC0) ? 2 : ((c & 0xF0) == 0xE0) ? 3 : 4;
    char tempChar[5] = {0};
    memcpy(tempChar, &text[i], charLen);
    i += charLen;

    if ((unsigned char)c < 0x80) {
      if (c == ' ' || c == '\n') {
        if (wordLen > 0) {
          if (lineWidth + wordWidth > maxWidth && lineLen > 0) {
            lineBuffer[lineLen] = '\0';
            u8g2.setCursor(startX, cursorY);
            u8g2.print(lineBuffer);
            cursorY += lineHeight;
            lineLen = 0; lineWidth = 0;
          }
          memcpy(lineBuffer + lineLen, wordBuffer, wordLen);
          lineLen += wordLen; lineWidth += wordWidth;
          wordLen = 0; wordWidth = 0;
        }
        if (c == ' ') {
          int spaceWidth = u8g2.getUTF8Width(" ");
          if (lineWidth + spaceWidth > maxWidth && lineLen > 0) {
            lineBuffer[lineLen] = '\0';
            u8g2.setCursor(startX, cursorY);
            u8g2.print(lineBuffer);
            cursorY += lineHeight;
            lineLen = 0; lineWidth = 0;
          } else {
            lineBuffer[lineLen++] = ' ';
            lineWidth += spaceWidth;
          }
        }
        if (c == '\n') {
          if (lineLen > 0) { lineBuffer[lineLen] = '\0'; u8g2.setCursor(startX, cursorY); u8g2.print(lineBuffer); }
          cursorY += lineHeight; lineLen = 0; lineWidth = 0;
        }
      } else {
        memcpy(wordBuffer + wordLen, tempChar, charLen);
        wordLen += charLen; wordBuffer[wordLen] = '\0';
        wordWidth = u8g2.getUTF8Width(wordBuffer);
      }
    }
  }
  if (wordLen > 0) {
    if (lineWidth + wordWidth > maxWidth && lineLen > 0) {
      lineBuffer[lineLen] = '\0'; u8g2.setCursor(startX, cursorY); u8g2.print(lineBuffer); cursorY += lineHeight; lineLen = 0;
    }
    memcpy(lineBuffer + lineLen, wordBuffer, wordLen);
    lineLen += wordLen;
  }
  if (lineLen > 0) {
    lineBuffer[lineLen] = '\0'; u8g2.setCursor(startX, cursorY); u8g2.print(lineBuffer);
  }
}

void writeWavHeader(uint8_t* buffer, int dataSize) {
  memcpy(buffer, "RIFF", 4);
  uint32_t chunkSize = dataSize + 36;
  memcpy(buffer + 4, &chunkSize, 4);
  memcpy(buffer + 8, "WAVEfmt ", 8);
  uint32_t subChunk1Size = 16;
  uint16_t audioFormat = 1;
  uint16_t numChannels = CHANNELS;
  uint32_t sampleRate = SAMPLE_RATE;
  uint32_t byteRate = sampleRate * CHANNELS * 2;
  uint16_t blockAlign = CHANNELS * 2;
  uint16_t bitsPerSample = 16;
  memcpy(buffer + 16, &subChunk1Size, 4);
  memcpy(buffer + 20, &audioFormat, 2);
  memcpy(buffer + 22, &numChannels, 2);
  memcpy(buffer + 24, &sampleRate, 4);
  memcpy(buffer + 28, &byteRate, 4);
  memcpy(buffer + 32, &blockAlign, 2);
  memcpy(buffer + 34, &bitsPerSample, 2);
  memcpy(buffer + 36, "data", 4);
  memcpy(buffer + 40, &dataSize, 4);
}

String sendToWhisper(uint8_t* wav_data, int wav_size) {
  client.setInsecure();
  if (!client.connect("api.openai.com", 443)) return "CONNECT FAIL";

  String boundary = "----ESP32Boundary7MA4YWxkTrZu0gW";
  String head = "--" + boundary + "\r\nContent-Disposition: form-data; name=\"model\"\r\n\r\n" + transcribe_model + "\r\n";
  head += "--" + boundary + "\r\nContent-Disposition: form-data; name=\"language\"\r\n\r\nen\r\n";
  head += "--" + boundary + "\r\nContent-Disposition: form-data; name=\"file\"; filename=\"audio.wav\"\r\nContent-Type: audio/wav\r\n\r\n";
  String tail = "\r\n--" + boundary + "--\r\n";
  int contentLength = head.length() + wav_size + tail.length();

  client.println("POST /v1/audio/transcriptions HTTP/1.1");
  client.println("Host: api.openai.com");
  client.println("Authorization: Bearer " + String(openai_key));
  client.println("Content-Type: multipart/form-data; boundary=" + boundary);
  client.println("Content-Length: " + String(contentLength));
  client.println("Connection: close");
  client.println();
  client.print(head);

  int sent = 0;
  const int chunkSize = 4096;
  while (sent < wav_size) {
    int toSend = min(chunkSize, wav_size - sent);
    client.write(wav_data + sent, toSend);
    sent += toSend;
  }
  client.print(tail);

  String response = "";
  unsigned long startTime = millis();
  while (client.connected() || client.available()) {
    if (client.available()) { response += client.readString(); startTime = millis(); }
    if (millis() - startTime > 8000) break;
  }
  client.stop();

  int jsonStart = response.indexOf("{");
  if (jsonStart < 0) return "NO JSON";
  DynamicJsonDocument doc(8192);
  if (deserializeJson(doc, response.substring(jsonStart))) return "JSON ERROR";
  return doc["text"].as<String>();
}

String askGPT(String question) {
  client.setInsecure();
  if (!client.connect("api.openai.com", 443)) return "CONNECT FAIL";

  DynamicJsonDocument doc(2048);
  doc["model"] = gpt_model;
  doc["max_tokens"] = 60;
  doc["temperature"] = 0.3;
  JsonArray messages = doc.createNestedArray("messages");
  JsonObject sys = messages.createNestedObject();
  sys["role"] = "system"; sys["content"] = "Answer briefly in 1-2 sentences.";
  JsonObject msg = messages.createNestedObject();
  msg["role"] = "user"; msg["content"] = question;

  String body; serializeJson(doc, body);
  client.println("POST /v1/chat/completions HTTP/1.1");
  client.println("Host: api.openai.com");
  client.println("Authorization: Bearer " + String(openai_key));
  client.println("Content-Type: application/json");
  client.println("Content-Length: " + String(body.length()));
  client.println("Connection: close");
  client.println();
  client.print(body);

  String response = "";
  while (client.connected() || client.available()) {
    if (client.available()) response += (char)client.read();
  }
  client.stop();

  int jsonStart = response.indexOf("{");
  if (jsonStart < 0) return "NO JSON";
  DynamicJsonDocument resp(4096);
  if (deserializeJson(resp, response.substring(jsonStart))) return "JSON ERROR";
  return resp["choices"][0]["message"]["content"].as<String>();
}

void recordClip() {
  int bytes_collected = 0;
  while (bytes_collected < RECORD_BYTES) {
    if (codecport != NULL) codecport->CodecPort_EchoRead(frame_buffer, AUDIO_FRAME_SIZE);
    int copy_bytes = min(AUDIO_FRAME_SIZE, RECORD_BYTES - bytes_collected);
    memcpy(record_buffer + bytes_collected, frame_buffer, copy_bytes);
    bytes_collected += copy_bytes;
  }
  writeWavHeader(wav_buffer, RECORD_BYTES);
  memcpy(wav_buffer + 44, record_buffer, RECORD_BYTES);
}

void readMicrophone() {
  if (codecport == NULL) return;
  codecport->CodecPort_EchoRead(frame_buffer, AUDIO_FRAME_SIZE);
  int16_t* pcm = (int16_t*)frame_buffer;
  int samples = AUDIO_FRAME_SIZE / 2;
  long long sum = 0;
  for (int i = 0; i < samples; i++) sum += (long long)pcm[i] * pcm[i];
  float rms = sqrt((float)sum / samples);
  if (rms > RMS_THRESHOLD) trigger_count++; else trigger_count = 0;
}

// ================= CONTROLLER =================
void handleButton() {
  bool reading = digitalRead(BUTTON_PIN);
  if (reading == LOW && lastButtonState == HIGH && (millis() - lastDebounceTime) > debounceDelay) {
    lastDebounceTime = millis();
    if (currentScreen == SCREEN_CLOCK) {
      currentScreen = SCREEN_BUS;
      updateWeatherAndClimate();
    } else if (currentScreen == SCREEN_BUS) {
      currentScreen = SCREEN_TRIVIA;
      updateBusArrival_one();
      updateBusArrival_two();
    } else if (currentScreen == SCREEN_TRIVIA) {
      currentScreen = SCREEN_GPT;
      draw_query();
    } else if (currentScreen == SCREEN_GPT) {
      currentScreen = SCREEN_CLOCK;
    }
  }
  drawScreen();
  lastButtonState = reading;
}

void handleScreenTimers() {
  unsigned long currentMillis = millis();
  if (currentScreen != previousScreen) {
    previousScreen = currentScreen;
    if (currentScreen == SCREEN_CLOCK) lastHomeUpdate = currentMillis;
    if (currentScreen == SCREEN_BUS)   lastBusUpdate = currentMillis;
  }

  if (currentScreen == SCREEN_CLOCK && (currentMillis - lastHomeUpdate >= SYNC_INTERVAL)) {
    lastHomeUpdate = currentMillis;
    connectWiFi();
    if (WiFi.status() == WL_CONNECTED) {
      read_batt_ADC();
      updateWeatherAndClimate();
      syncTimeWithNTP();
    }
    WiFi.disconnect(true);
    WiFi.mode(WIFI_OFF);
  }

  if (currentScreen == SCREEN_BUS && (currentMillis - lastBusUpdate >= BUS_INTERVAL)) {
    lastBusUpdate = currentMillis;
    updateBusArrival_one();
    updateBusArrival_two();
  }
}

void drawScreen() {
  switch (currentScreen) {
    case SCREEN_CLOCK:
      updateClockDisplay();
      break;

    case SCREEN_BUS:
      number_of_bus_page_one = (number_of_bus_one + chunkSize - 1) / chunkSize;
      number_of_bus_page_two = (number_of_bus_two + chunkSize - 1) / chunkSize;      
      if (millis() - lastBusScreen >= BusScreen_INTERVAL) {
        lastBusScreen = millis();
        Current_Page_one = (Current_Page_one >= number_of_bus_page_one) ? 1 : Current_Page_one + 1;
        chunk_one_length = (Current_Page_one == number_of_bus_page_one && (number_of_bus_one % chunkSize != 0)) 
                            ? (number_of_bus_one % chunkSize) : chunkSize;
        for (int i = 0; i < chunkSize; i++) chunk_one[i] = data_one[(Current_Page_one - 1) * chunkSize + i];

        Current_Page_two = (Current_Page_two >= number_of_bus_page_two) ? 1 : Current_Page_two + 1;
        chunk_two_length = (Current_Page_two == number_of_bus_page_two && (number_of_bus_two % chunkSize != 0)) 
                            ? (number_of_bus_two % chunkSize) : chunkSize;
        for (int i = 0; i < chunkSize; i++) chunk_two[i] = data_two[(Current_Page_two - 1) * chunkSize + i];
      }
      drawBus();
      break;

    case SCREEN_TRIVIA:
      switch (TRIVIAL_currentScreen) {
        case IMAGE_STATE:    handleImage(); break;
        case QUESTION_STATE: handleTrivia_question(); break;
        case ANSWER_STATE:   drawTrivial_answer(); break;
      }
      break;

    case SCREEN_GPT:
      if (state == IDLE) readMicrophone();
      switch (state) {
        case IDLE:
          if (trigger_count >= TRIGGER_FRAMES) { trigger_count = 0; state = RECORDING; }
          break;
        case RECORDING:
          recordClip();
          state = PROCESSING;
          break;
        case PROCESSING:
          handleProcessing();
          break;
        case DISPLAY_RESPONSE:
          handleResponse();
          break;
      }
      break;
  }
}

// ================= SETUP & LOOP =================
void setup() {
  Serial.begin(115200);
  delay(200);

  analogReadResolution(12);
  analogSetPinAttenuation(BAT_ADC_PIN, ADC_11db);

  get_sdcard_config();
  read_batt_ADC();

  pinMode(BUTTON_PIN, INPUT_PULLUP);
  RlcdPort.RLCD_Init();

  /* ---- Audio Hardware Power Enable ---- */
  pinMode(PA_CTRL, OUTPUT);
  digitalWrite(PA_CTRL, HIGH);
  delay(500); 

  record_buffer = (uint8_t*)heap_caps_malloc(RECORD_BYTES, MALLOC_CAP_SPIRAM);
  wav_buffer    = (uint8_t*)heap_caps_malloc(RECORD_BYTES + 44, MALLOC_CAP_SPIRAM);

  // Initialize Codec & RTC using the single shared I2cbus
  codecport = new CodecPort(I2cbus, "S3_RLCD_4_2");
  codecport->CodecPort_SetInfo("es8311 & es7210", 1, SAMPLE_RATE, CHANNELS, 16);
  codecport->CodecPort_SetMicGain(35);

  rtc.setBus(I2cbus);
  rtc.begin();

  // Initial SHTC3 read over I2cbus
  shtc3_read_i2cbus(temp, humidity);

  connectWiFi();
  updateWeatherAndClimate();
  updateBusArrival_one();
  updateBusArrival_two();

  drawScreen();
}

void loop() {
  handleButton();
  handleScreenTimers();

  // Read internal SHTC3 sensor & battery every 200 cycles (~10 seconds)
  if (sensor_counter == 0) {
    read_batt_ADC();
    if (shtc3_read_i2cbus(temp, humidity)) {
      Serial.printf("SHTC3 Sensor: T=%.1f C, RH=%.1f %%\n", temp, humidity);
    } else {
      Serial.println("SHTC3 read failed");
    }
  }

  sensor_counter++;
  if (sensor_counter >= 200) sensor_counter = 0;

  delay(50); // Frame throttle prevents SPI/I2C bus overload and CPU starvation
}
